def collect_risk_input() -> dict:
    """汇总 risk.rego 的 input（复用 affected + 实填 boundary_violations）。"""
    data = collect_affected()
    data["boundary_violations"] = _count_boundary_violations(data["changed_files"])
    return data


def _count_boundary_violations(changed_files: list[str], specs_dir: Path | None = None) -> int:
    allowed = _collect_all_allowed(specs_dir)
    n = 0
    for f in changed_files:
        f = f.replace("\\", "/")
        if f in allowed or _is_whitelisted(f):
            continue
        n += 1
    return n


def _collect_all_allowed(specs_dir: Path | None = None) -> set[str]:
    specs_dir = specs_dir or SPECS_DIR
    allowed: set[str] = set()
    for spec in Path(specs_dir).glob("*.spec.md"):
        try:
            contract = parse_contract_file(spec)
        except Exception:
            continue
        allowed.update(p.replace("\\", "/") for p in contract.allowed_changes)
    return allowed


def _is_whitelisted(path: str) -> bool:
    if any(path.startswith(d) for d in WHITELIST_DIRS):
        return True
    if "/" not in path and (path in WHITELIST_FILES or path.endswith(".md")):
        return True
    return False
```

def _append_shadow(result: dict, input_data: dict, log_dir: Path | None = None) -> None:
    """追加影子记录到 .out/shadow.jsonl（L1→L1.5 切换准入用）。"""
    import subprocess
    import time
    log_dir = log_dir or (POLICY_DIR.parent / ".out")
    log_dir.mkdir(parents=True, exist_ok=True)
    commit = subprocess.run(
        ["git", "rev-parse", "--short=10", "HEAD"],
        capture_output=True, text=True).stdout.strip()
    record = {
        "ts": int(time.time()),
        "commit": commit,
        "level": result.get("level"),
        "deny": result.get("deny", []),
        "changed_files": input_data.get("changed_files", []),
        "human_decision": None,  # PR 合并时回填（阶段后续）
    }
    with (log_dir / "shadow.jsonl").open("a", encoding="utf-8") as f:
        f.write(json.dumps(record, ensure_ascii=False) + "\n")
```

          python -m spec_runner risk > .out/risk.json
          cat .out/risk.json
          python -c "import json,sys; d=json.load(open('.out/risk.json')); sys.exit(1 if d['deny'] else 0)"

- [ ] **Step 2: 新增 auto-merge job**

在 `verify.yml` 末尾（contract job 之后）加：

```yaml
  auto-merge:
    needs: contract
    if: github.event_name == 'pull_request' && needs.contract.result == 'success'
    runs-on: ubuntu-latest
    permissions:
      contents: write
      pull-requests: write
    steps:
      - name: R0 自动合并
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          PR_NUMBER: ${{ github.event.pull_request.number }}
          LEVEL: ${{ needs.contract.outputs.risk-level }}
        run: |
          echo "risk level: $LEVEL"
          if [ "$LEVEL" = "R0" ]; then
            gh pr merge "$PR_NUMBER" --auto --squash --delete-branch
            echo "R0 → 自动合并（squash）"
          else
            gh pr edit "$PR_NUMBER" --add-label "needs-review"
            echo "非 R0 → 标 needs-review，等人审"
          fi
```
